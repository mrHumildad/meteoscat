# Geo tooling for meteokat/build_lithology.py (grid generator).
#
# These are ONLY needed on the machine that rebuilds public/logic/litho_grid.json
# from the ICGC GeoPackage — not at app runtime, and not by the daily scraper.
#
#     python3 -m venv meteokat/venv
#     meteokat/venv/bin/pip install -r meteokat/requirements-build.txt
#
fiona          # read the ICGC GeoPackage (unit polygons)
rasterio       # rasterize the polygons onto the lon/lat grid
pyproj         # reproject EPSG:25831 -> EPSG:4326
shapely        # geometry handling (audit)
